cmake_minimum_required(VERSION 3.18)
project(dvb_ts_regression LANGUAGES CXX)
enable_testing()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
get_filename_component(SATTOOL_ROOT ../.. ABSOLUTE)
add_executable(dvb_ts_regression regression.cpp
    ${SATTOOL_ROOT}/src-core/common/codings/dvb-s2/bbframe_ts_parser.cpp)
target_include_directories(dvb_ts_regression PRIVATE ${SATTOOL_ROOT}/src-core)
target_link_libraries(dvb_ts_regression PRIVATE Threads::Threads)
if(WIN32)
    target_link_libraries(dvb_ts_regression PRIVATE ws2_32)
endif()
add_test(NAME dvb_ts_regression COMMAND dvb_ts_regression)
set_tests_properties(dvb_ts_regression PROPERTIES TIMEOUT 30)
